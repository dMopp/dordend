# rummage's preview helpers (ROADMAP 2b): the client side, linked into rummage, and the
# rummage-preview program that runs the parsers.
add_library(rummage_isolate
    src/helpers.cpp
    src/protocol.cpp)
add_library(rummage::isolate ALIAS rummage_isolate)
target_include_directories(rummage_isolate PUBLIC include PRIVATE src)
target_link_libraries(rummage_isolate PUBLIC rummage::picture rummage::text rummage::engines
                                      PRIVATE rummage::core rummage::warnings)

# Tiny pictures of every kind the helper reads, decoded once before it closes its cage, so that
# the libraries load what they need while they still can: generated from the test fixtures at
# configure time (pure CMake, no tool needed).
set(RUMMAGE_WARM_UP_FILES blocks.png blocks.gif blocks.jpg shot.jpg with-thumbnail.jpg
    blocks.webp blocks-anim.webp blocks.jxl tiff-preview.tif)
set(warm_up_source "${CMAKE_CURRENT_BINARY_DIR}/warm_up_data.cpp")
set(warm_up_text "// Generated by src/isolate/CMakeLists.txt from tests/data/image. Do not edit.\n")
string(APPEND warm_up_text "#include \"warm_up.hpp\"\nnamespace rummage::isolate {\n")
set(warm_up_list "")
set(index 0)
foreach(name IN LISTS RUMMAGE_WARM_UP_FILES)
    set(fixture "${PROJECT_SOURCE_DIR}/tests/data/image/${name}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${fixture}")
    file(READ "${fixture}" hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
    string(APPEND warm_up_text "static const unsigned char warm_up_${index}[] = {${bytes}};\n")
    string(APPEND warm_up_list "{\"${name}\", warm_up_${index}, sizeof(warm_up_${index})},")
    math(EXPR index "${index} + 1")
endforeach()
string(APPEND warm_up_text "const WarmUpPicture kWarmUp[] = {${warm_up_list}};\n")
string(APPEND warm_up_text "const std::size_t kWarmUpCount = ${index};\n}\n")
file(CONFIGURE OUTPUT "${warm_up_source}" CONTENT "${warm_up_text}")

add_executable(rummage-preview src/base_backend.cpp src/helper_main.cpp src/protocol.cpp src/sandbox.cpp
                               "${warm_up_source}")
target_include_directories(rummage-preview PRIVATE src)
# Generated, not ours to lint.
set_source_files_properties("${warm_up_source}" PROPERTIES SKIP_LINTING ON)
target_link_libraries(rummage-preview
    PRIVATE rummage::formats rummage::formats::text rummage::picture rummage::text rummage::warnings)
# Next to rummage in the build tree, where default_program() looks first.
set_target_properties(rummage-preview PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
